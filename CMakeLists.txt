cmake_minimum_required(VERSION 3.24)
project(FrescoCAD VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Test)
qt_standard_project_setup(REQUIRES 6.8)
add_library(cad_core src/document.cpp src/document.h)
target_include_directories(cad_core PUBLIC src)
target_link_libraries(cad_core PUBLIC Qt6::Core)
qt_add_executable(fresco_cad MACOSX_BUNDLE src/main.cpp src/canvas.cpp src/canvas.h src/geometry.h)
target_link_libraries(fresco_cad PRIVATE cad_core Qt6::Quick Qt6::QuickControls2)
qt_add_resources(fresco_cad app_resources PREFIX "/" FILES qml/Main.qml qml/CadButton.qml qml/Field.qml qml/theme.json qml/strings.json)
set_target_properties(fresco_cad PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Fresco CAD" MACOSX_BUNDLE_GUI_IDENTIFIER "internal.fresco.cad.spike")
include(CTest)
if(BUILD_TESTING)
qt_add_executable(core_check tests/core_check.cpp)
target_link_libraries(core_check PRIVATE cad_core)
add_test(NAME core_check COMMAND core_check)
qt_add_executable(canvas_check tests/canvas_check.cpp src/canvas.cpp src/canvas.h src/geometry.h)
target_link_libraries(canvas_check PRIVATE cad_core Qt6::Quick Qt6::QuickControls2 Qt6::Test)
qt_add_resources(canvas_check check_resources PREFIX "/" FILES qml/Field.qml qml/theme.json)
add_test(NAME canvas_check COMMAND canvas_check)
set_tests_properties(canvas_check PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
qt_add_executable(render_check tests/render_check.cpp src/canvas.cpp src/canvas.h src/geometry.h)
target_link_libraries(render_check PRIVATE cad_core Qt6::Quick Qt6::QuickControls2)
add_test(NAME render_check COMMAND render_check)
set_tests_properties(render_check PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
if(APPLE)
qt_add_executable(render_visual_check tests/render_visual_check.cpp src/canvas.cpp src/canvas.h src/geometry.h)
target_link_libraries(render_visual_check PRIVATE cad_core Qt6::Quick Qt6::QuickControls2 Qt6::Test)
endif()
endif()
install(TARGETS fresco_cad BUNDLE DESTINATION . RUNTIME DESTINATION bin)
